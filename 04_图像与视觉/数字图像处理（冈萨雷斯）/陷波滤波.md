def remove_periodic_noise(img, spikes, R=5):
    H, W = img.shape
    pad_img = np.pad(img, ((0,H),(0,W)))
    F = fft.fft2(pad_img)
    Fshift = fft.fftshift(F)
    H_notch = np.ones_like(Fshift)
    
    cy, cx = H, W
    for u0, v0 in spikes:
        y, x = np.ogrid[0:2*H, 0:2*W]
        D1 = np.sqrt((x - cx - u0)**2 + (y - cy - v0)**2)
        D2 = np.sqrt((x - cx + u0)**2 + (y - cy + v0)**2)
        H_notch[(D1 <= R) | (D2 <= R)] = 0
    
    G = H_notch * Fshift
    g = np.real(fft.ifft2(fft.ifftshift(G)))
    return g[:H, :W]
