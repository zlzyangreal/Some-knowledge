```python
def estimate_noise_variance(img, patch_size=15, num_patches=10):
    """在图像平坦区域估计噪声方差"""
    h, w = img.shape
    variances = []
    for _ in range(num_patches):
        y = np.random.randint(patch_size, h - patch_size)
        x = np.random.randint(patch_size, w - patch_size)
        patch = img[y-patch_size//2:y+patch_size//2+1,
                    x-patch_size//2:x+patch_size//2+1]
        variances.append(np.var(patch))
    return np.median(variances)

# 使用
img = cv2.imread('noisy.jpg', 0).astype(np.float32) / 255.0
noise_var = estimate_noise_variance(img)
K = 10 * noise_var  # 维纳滤波常数
print(f"Estimated noise variance: {noise_var:.6f} → K = {K:.4f}")
```
- `K` 控制复原锐度与噪声抑制平衡
