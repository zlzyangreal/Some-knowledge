# 3. Code
```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def geometric_mean_filter(img, kernel, a=0.5, K=0.01):
    """
    几何均值滤波复原
    img    : 退化图像 (float32, 0-1)
    kernel : PSF 核 (已归一化)
    a      : 锐化指数 (0~1)
    K      : 噪声功率常数
    """
    h, w = img.shape
    kh, kw = kernel.shape
    
    # 零填充
    pad_h = h + kh - 1
    pad_w = w + kw - 1
    img_pad = np.pad(img, ((0, pad_h-h), (0, pad_w-w)), 'constant')
    kernel_pad = np.pad(kernel, ((0, pad_h-kh), (0, pad_w-kw)), 'constant')
    
    # 频域运算
    G = np.fft.fft2(img_pad)
    H = np.fft.fft2(kernel_pad)
    H_conj = np.conj(H)
    H_mag = np.abs(H)
    
    # 防止除零
    H_mag_safe = np.where(H_mag < 1e-8, 1e-8, H_mag)
    
    # 几何均值滤波器
    W = H_conj / (H_mag_safe**(2*a) * K**(1-a))
    
    # 复原
    F_restored = G * W
    result = np.real(np.fft.ifft2(F_restored))
    return result[:h, :w]
```
