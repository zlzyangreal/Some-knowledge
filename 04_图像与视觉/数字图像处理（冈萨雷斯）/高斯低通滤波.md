def denoise_lowpass(img, D0=40):
    H, W = img.shape
    # 1. 零填充防 wrap-around
    pad_img = np.pad(img, ((0, H), (0, W)), mode='constant')
    
    # 2. 构造高斯低通
    center_y, center_x = H, W  # 填充后中心
    y, x = np.ogrid[0:2*H, 0:2*W]
    D = np.sqrt((x - center_x)**2 + (y - center_y)**2)
    H_filter = np.exp(-D**2 / (2 * D0**2))
    
    # 3. 频域相乘
    F = fft.fft2(pad_img)
    G = H_filter * fft.fftshift(F)
    
    # 4. IFFT + 裁剪
    g = np.real(fft.ifft2(fft.ifftshift(G)))
    return g[:H, :W]

# === 直接跑 ===
img = load_img('noisy.jpg')
clean = denoise_lowpass(img, D0=40)
show(img, clean, spectrum(np.pad(img, ((0,img.shape[0]),(0,img.shape[1])))), 
     spectrum(np.pad(clean, ((0,clean.shape[0]),(0,clean.shape[1])))),
     titles=['原图', '去噪后', '原频谱', '去噪频谱'])
