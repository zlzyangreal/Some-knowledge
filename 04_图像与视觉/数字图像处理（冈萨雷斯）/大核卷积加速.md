```python
def conv_freq(img, kernel):
    kh, kw = kernel.shape
    # 零填充到 img + kernel - 1
    pad_h = img.shape[0] + kh - 1
    pad_w = img.shape[1] + kw - 1
    img_pad = np.pad(img, ((0, pad_h - img.shape[0]), (0, 2*pad_w - img.shape[1])), 'constant')
    kernel_pad = np.pad(kernel, ((0, pad_h - kh), (0, pad_w - kw)), 'constant')
    
    F = fft.fft2(img_pad)
    H = fft.fft2(kernel_pad)
    G = F * H
    result = np.real(fft.ifft2(G))
    return result[:img.shape[0], :img.shape[1]]

# === 造 101×101 高斯核 ===
size = 101
sigma = 30
x = np.arange(-size//2, size//2 + 1)
X, Y = np.meshgrid(x, x)
kernel = np.exp(-(X**2 + Y**2)/(2*sigma**2))
kernel /= kernel.sum()

blurred = conv_freq(img, kernel)
show(img, blurred, titles=['原图', '101×101 高斯（频域）'])
```
