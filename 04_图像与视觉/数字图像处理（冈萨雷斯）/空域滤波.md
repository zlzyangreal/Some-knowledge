```python
import cv2
import os

def median_filter_gray(input_path, output_path=None, ksize=3):
    """
    对灰度图像进行中值滤波去噪。
    
    参数:
        input_path: 输入图像路径（灰度图或彩色图均可）
        output_path: 输出图像路径（可选）
                     若未指定，自动在同目录生成 *_median.jpg
        ksize: 滤波核大小（必须是奇数，典型值为3、5、7）
    
    返回:
        输出图像保存路径
    """
    # 读取图像（强制转灰度）
    img = cv2.imread(input_path, cv2.IMREAD_GRAYSCALE)
    if img is None:
        raise FileNotFoundError(f"无法读取图像：{input_path}")

    # 检查核大小
    if ksize % 2 == 0 or ksize < 1:
        raise ValueError("ksize 必须是奇数且大于 0，例如 3、5、7")

    # 执行中值滤波
    filtered = cv2.medianBlur(img, ksize)

    # 自动命名输出路径
    if output_path is None:
        base, ext = os.path.splitext(input_path)
        output_path = f"{base}_median{ext}"

    # 保存结果
    cv2.imwrite(output_path, filtered)
    return output_path


# ===== 示例调用 =====
if __name__ == "__main__":
    result = median_filter_gray("input_gray.jpg", ksize=5)
    print(f"中值滤波结果已保存：{result}")
```
