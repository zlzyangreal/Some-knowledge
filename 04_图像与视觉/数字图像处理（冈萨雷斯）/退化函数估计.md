# 运动模糊参数估计（频谱零点法）
def estimate_motion_length(spec_log):
    """spec_log: log频谱图"""
    plt.imshow(spec_log, cmap='gray')
    plt.title("Click two ends of a dark stripe to estimate L")
    pts = plt.ginput(2)
    plt.close()
    dist = np.linalg.norm(np.array(pts[0]) - np.array(pts[1]))
    L = int(spec_log.shape[0] / dist)
    return L
