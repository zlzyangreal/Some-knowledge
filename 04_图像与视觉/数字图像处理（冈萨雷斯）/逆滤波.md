```python
def inverse_filter(img, kernel, epsilon=1e-6):
    """
    逆滤波复原
    img     : 退化图像 (float32, 0-1)
    kernel  : PSF 核 (已归一化)
    epsilon : 防除零小常数
    """
    h, w = img.shape
    kh, kw = kernel.shape
    
    # 零填充至避免边界效应
    pad_h = h + kh - 1
    pad_w = w + kw - 1
    img_pad = np.pad(img, ((0, pad_h-h), (0, pad_w-w)), 'constant')
    kernel_pad = np.pad(kernel, ((0, pad_h-kh), (0, pad_w-kw)), 'constant')
    
    # 频域运算
    G = np.fft.fft2(img_pad)
    H = np.fft.fft2(kernel_pad)
    
    # 逆滤波：直接除以 H，防止除零
    H_safe = H.copy()
    H_safe[np.abs(H) < epsilon] = epsilon  # 避免除零
    F_restored = G / H_safe
    
    # IFFT 取实部 + 裁剪
    result = np.real(np.fft.ifft2(F_restored))
    return result[:h, :w]
```
