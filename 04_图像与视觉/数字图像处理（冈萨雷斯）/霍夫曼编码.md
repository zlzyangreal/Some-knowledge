根据符号出现频率，分配“高频短码、低频长码”的变长码字，实现无损压缩。
# 1. 作用
![[图像压缩3.png]]
把图像数据从“固定8位/像素”变成“按需编码”，在无损前提下最大化压缩比。
# 2. 怎么使用
1. **统计频率**：数每个像素值出现次数
2. **建最小堆**：频率从小到大排队
3. **合并节点**：每次取两个最小，合并为父节点（频率相加）
4. **建树**：重复合并直到剩一个根
5. **生成码表**：左=0，右=1，从根到叶子
6. **编码/解码**：
    - **编码**：像素 → 码字 → 比特流 → 字节
    - **解码**：比特流沿树走，匹配叶子输出像素
# 3.Code
```python
import numpy as np
from collections import Counter
import heapq
import struct

# ------------------- 霍夫曼树节点 -------------------
class HuffmanNode:
    def __init__(self, symbol=None, freq=0, left=None, right=None):
        self.symbol = symbol
        self.freq = freq
        self.left = left
        self.right = right

    def __lt__(self, other):
        return self.freq < other.freq

# ------------------- 构建树 + 码表 -------------------
def build_huffman_tree(pixel_data):
    freq = Counter(pixel_data)
    heap = [HuffmanNode(s, f) for s, f in freq.items()]
    heapq.heapify(heap)
    while len(heap) > 1:
        left = heapq.heappop(heap)
        right = heapq.heappop(heap)
        merged = HuffmanNode(freq=left.freq + right.freq, left=left, right=right)
        heapq.heappush(heap, merged)
    return heap[0]

def generate_codes(node, current_code="", codebook=None):
    if codebook is None: codebook = {}
    if node.symbol is not None:
        codebook[node.symbol] = current_code if current_code else "0"
        return codebook
    generate_codes(node.left, current_code + "0", codebook)
    generate_codes(node.right, current_code + "1", codebook)
    return codebook

# ------------------- 编码 -------------------
def huffman_encode(img):
    img_flat = img.flatten()
    root = build_huffman_tree(img_flat)
    codebook = generate_codes(root)

    encoded_bits = ''.join(codebook[pixel] for pixel in img_flat)
    padding = 8 - len(encoded_bits) % 8
    encoded_bits += '0' * padding
    byte_data = bytearray(int(encoded_bits[i:i+8], 2) for i in range(0, len(encoded_bits), 8))

    return bytes(byte_data), codebook, padding, img.shape

# ------------------- 解码 -------------------
def huffman_decode(encoded_bytes, codebook, padding, shape):
    reverse_codebook = {v: k for k, v in codebook.items()}
    bit_string = ''.join(bin(b)[2:].zfill(8) for b in encoded_bytes)[:-padding]

    decoded_pixels = []
    current_code = ""
    for bit in bit_string:
        current_code += bit
        if current_code in reverse_codebook:
            decoded_pixels.append(reverse_codebook[current_code])
            current_code = ""

    return np.array(decoded_pixels).reshape(shape)
```
## 3.1 使用实例
```python
img = np.array([[100, 100, 101, 102],
                [100, 101, 102, 102],
                [103, 103, 103, 104],
                [104, 104, 105, 105]], dtype=np.uint8)

encoded_data, codebook, padding, shape = huffman_encode(img)
print(f"压缩比: {img.size / len(encoded_data):.2f}:1")

decoded_img = huffman_decode(encoded_data, codebook, padding, shape)
print("无损还原:", np.array_equal(img, decoded_img))
```
- 存文件
```python
with open('img.huff', 'wb') as f:
    f.write(struct.pack('IIB', *shape, padding) + encoded_data)
```
