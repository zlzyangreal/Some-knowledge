# 1. 代码展示
```python
import cv2
import numpy as np
import os

def high_boost_filter_gray(input_path, output_path=None, A=1.5, blur_ksize=5):
    """
    对灰度图像进行高提升滤波以增强细节。
    
    参数:
        input_path: 输入图像路径（灰度或彩色图都可以）
        output_path: 输出图像路径（可选），默认生成 *_highboost.jpg
        A: 增强系数（>1）。1.0表示普通锐化，1.3~2.0适中增强，>2则增强更强
        blur_ksize: 平滑核大小（建议奇数，如 3、5、7）
    
    返回:
        输出图像保存路径
    """
    # 读取图像（灰度）
    img = cv2.imread(input_path, cv2.IMREAD_GRAYSCALE)
    if img is None:
        raise FileNotFoundError(f"无法读取图像：{input_path}")

    # 平滑（低通滤波）
    low_pass = cv2.GaussianBlur(img, (blur_ksize, blur_ksize), 0)

    # 高通分量
    mask = cv2.subtract(img, low_pass)

    # 高提升滤波结果
    high_boost = cv2.addWeighted(img, A, mask, 1.0, 0)

    # 自动命名输出路径
    if output_path is None:
        base, ext = os.path.splitext(input_path)
        output_path = f"{base}_highboost{ext}"

    # 保存结果
    cv2.imwrite(output_path, high_boost)
    return output_path


# ===== 示例调用 =====
if __name__ == "__main__":
    result = high_boost_filter_gray("input_gray.jpg", A=1.8, blur_ksize=5)
    print(f"高提升滤波图已保存：{result}")
```
