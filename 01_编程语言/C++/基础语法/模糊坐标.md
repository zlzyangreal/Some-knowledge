#include <iostream>
#include <string>
#include <vector>
#include <algorithm>
#include <cctype> // for isdigit

class AmbiguousCoordinates {
private:
    std::vector<int> arr_;

    // [新增] 私有辅助函数：生成一段数字区间内所有合法的字符串（整数或小数）
    std::vector<std::string> generatePart(int start, int end);

public:
    AmbiguousCoordinates(const std::string& str);
    
    // [修改] 返回值改为 string vector，因为结果包含 "1.23" 这种格式
    std::vector<std::string> GetResults();
};

// 1. 构造函数优化：更稳健的数字提取
AmbiguousCoordinates::AmbiguousCoordinates(const std::string& str) {
    // 预分配内存，避免频繁扩容
    arr_.reserve(str.size());
    
    // 简单的状态机或直接遍历：只提取数字
    // 题目通常输入是 "(123)" 或 "(1, 23)"，我们只需要里面的数字序列
    for (char c : str) {
        if (std::isdigit(c)) {
            arr_.push_back(c - '0');
        }
    }
}

// 2. [核心逻辑] 生成合法数字（处理小数点和前导/后置零）
std::vector<std::string> AmbiguousCoordinates::generatePart(int start, int end) {
    std::vector<std::string> res;
    std::string s = "";
    
    // 将 vector 片段转为纯字符串，例如 {1, 2, 3} -> "123"
    for (int k = start; k < end; ++k) {
        s += std::to_string(arr_[k]);
    }
    
    int len = s.length();

    // 情况 A: 不加小数点（纯整数）
    // 合法规则：长度为1，或者首位不是 '0'
    if (len == 1 || s[0] != '0') {
        res.push_back(s);
    }

    // 情况 B: 尝试插入小数点
    // k 代表整数部分的长度，从 1 到 len-1
    if (len > 1) {
        for (int k = 1; k < len; ++k) {
            std::string left = s.substr(0, k);
            std::string right = s.substr(k);

            // 规则1：整数部分 (left) 不能以 0 开头（除非它是单独的 "0"）
            bool leftValid = (left.length() == 1 || left[0] != '0');
            
            // 规则2：小数部分 (right) 末尾不能是 0
            bool rightValid = (right.back() != '0');

            if (leftValid && rightValid) {
                res.push_back(left + "." + right);
            }
        }
    }
    
    return res;
}

// 3. 主逻辑：切分左右两部分并组合
std::vector<std::string> AmbiguousCoordinates::GetResults() {
    std::vector<std::string> result;
    int size = arr_.size();
    if (size < 2) return result;

    // 循环切分点 i (左半部分的长度)
    for (int i = 1; i < size; ++i) {
        // 分别生成左右两边所有合法的形式
        // 左边范围: [0, i)
        std::vector<std::string> leftList = generatePart(0, i);
        
        // 右边范围: [i, size)
        std::vector<std::string> rightList = generatePart(i, size);

        // 如果任一边无法生成合法数字，则当前切分无效
        if (leftList.empty() || rightList.empty()) continue;

        // 笛卡尔积：组合左边和右边的所有结果
        for (const std::string& l : leftList) {
            for (const std::string& r : rightList) {
                result.push_back("(" + l + ", " + r + ")");
            }
        }
    }
    return result;
}

int main() {
    std::string s;
    if (std::getline(std::cin, s)) {
        AmbiguousCoordinates ac(s);
        auto results = ac.GetResults();
        for (const auto& r : results) {
            std::cout << r << "\n";
        }
    }
    return 0;
}
