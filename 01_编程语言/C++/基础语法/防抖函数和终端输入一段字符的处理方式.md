#include <iostream>
#include <string>
#include <vector>
#include <algorithm>
#include <sstream>

#ifdef _WIN32
    #include <conio.h>
#else
    #include <termios.h>
    #include <unistd.h>
#endif


class AntiShake {
public:
    AntiShake();

    void AntiShakeresult();
private:
    struct Node_ {
        int t = 0;
        std::vector<int> inputs;
    };

    int t_;
    std::vector<Node_> nodes_;
    std::vector<Node_> result_;

    std::string readUnbufferedInput_(); 
    void parseInput_(std::string input); 
    

    std::string extractBetween_(const std::string& src, const std::string& start, const std::string& end, size_t& cursor) const;

    int safeToInt_(const std::string& s) const;

    void printResult_() const;
};



AntiShake::AntiShake() : t_(0) {
    std::string input = readUnbufferedInput_();
    
    parseInput_(input);
}

void AntiShake:: printResult_() const {
        std::cout << "--- AntiShake Result ---" << std::endl;
        for (const auto& node : result_) {
            std::cout << "t=" << node.t << " calls=[";
            for (size_t j = 0; j < node.inputs.size(); ++j) {
                std::cout << node.inputs[j] << (j == node.inputs.size() - 1 ? "" : ", ");
            }
            std::cout << "]" << std::endl;
        }
    }

void AntiShake::AntiShakeresult() {
    if (nodes_.empty()) return;

    result_.clear();

    Node_ pending_node = nodes_[0];

    for (size_t i = 1; i < nodes_.size(); ++i) {
        const Node_& current_node = nodes_[i];

        int pending_finish_time = pending_node.t + t_;

        if (current_node.t <= pending_finish_time) {
            pending_node = current_node; 
        } else {

            Node_ res;
            res.t = pending_finish_time;     
            res.inputs = pending_node.inputs; 
            result_.push_back(res);

            pending_node = current_node;
        }
    }


    Node_ last_res;
    last_res.t = pending_node.t + t_;
    last_res.inputs = pending_node.inputs; 
    result_.push_back(last_res);
    printResult_();
}
int AntiShake::safeToInt_(const std::string& s) const {
    if (s.empty()) return 0;
    return std::atoi(s.c_str());
}

std::string AntiShake::extractBetween_(const std::string& src, const std::string& start, const std::string& end, size_t& cursor) const {
    size_t s_pos = src.find(start, cursor);
    if (s_pos == std::string::npos) return "";
    
    s_pos += start.length();
    size_t e_pos = src.find(end, s_pos);
    if (e_pos == std::string::npos) return "";

    cursor = e_pos;
    return src.substr(s_pos, e_pos - s_pos);
}

void AntiShake::parseInput_(std::string input) {
    if (input.empty()) return;


    input.erase(std::remove_if(input.begin(), input.end(), ::isspace), input.end());

    size_t cursor = 0;

    std::string t_str = extractBetween_(input, "t=", "calls=[", cursor);
    if (!t_str.empty()) {
        t_ = safeToInt_(t_str);
    }

    size_t array_end = input.find_last_of(']');
    if (array_end == std::string::npos) return;

    while (cursor < array_end) {
        size_t obj_start = input.find('{', cursor);
        if (obj_start == std::string::npos || obj_start >= array_end) break;

        size_t temp_cursor = obj_start;
        std::string obj_content = extractBetween_(input, "{", "}", temp_cursor);
        cursor = temp_cursor; 

        if (obj_content.empty()) continue;

        Node_ node;
        size_t inner_cursor = 0;

        std::string node_t_str = extractBetween_(obj_content, "\"t\":", ",", inner_cursor);
        
        if (node_t_str.empty()) {
             size_t key_pos = obj_content.find("\"t\":");
             if (key_pos != std::string::npos) {
                 node.t = safeToInt_(obj_content.substr(key_pos + 4));
             }
        } else {
            node.t = safeToInt_(node_t_str);
        }

        size_t inputs_pos = obj_content.find("inputs:[");
        if (inputs_pos != std::string::npos) {
            size_t inputs_end = obj_content.find(']', inputs_pos);
            if (inputs_end != std::string::npos) {
                std::string nums = obj_content.substr(inputs_pos + 8, inputs_end - (inputs_pos + 8));
                std::stringstream ss(nums);
                std::string segment;
                while (std::getline(ss, segment, ',')) {
                    if (!segment.empty()) {
                        node.inputs.push_back(safeToInt_(segment));
                    }
                }
            }
        }
        nodes_.push_back(node);
    }
}

std::string AntiShake::readUnbufferedInput_() {
    std::string result;
    
#ifdef _WIN32
    char ch;
    std::cout << "Enter characters (press ESC to finish): " << std::flush;
    
    while (true) {
        if (_kbhit()) {
            ch = _getch();
            
            if (ch == 27) { // ESC
                break;
            } else if (ch == 8 || ch == 127) { // Backspace
                if (!result.empty()) {
                    result.pop_back();
                    std::cout << "\b \b";
                }
            } else if (ch >= 32 && ch <= 126) {
                result += ch;
                std::cout << ch << std::flush;
            }
        }
    }
#else
    struct termios old_term, new_term;
    tcgetattr(STDIN_FILENO, &old_term);
    new_term = old_term;
    new_term.c_lflag &= ~(ICANON | ECHO);
    tcsetattr(STDIN_FILENO, TCSANOW, &new_term);
    
    std::cout << "Enter characters (press ESC to finish): " << std::flush;
    
    char ch;
    while (true) {
        ch = getchar();
        
        if (ch == 27) {
            break;
        } else if (ch == 127 || ch == 8) {
            if (!result.empty()) {
                result.pop_back();
                std::cout << "\b \b" << std::flush;
            }
        } else if (ch >= 32 && ch <= 126) {
            result += ch;
            std::cout << ch << std::flush;
        }
    }
    tcsetattr(STDIN_FILENO, TCSANOW, &old_term);
#endif
    
    std::cout << std::endl;
    return result;
}


int main() {
    AntiShake as;
