#include <iostream>
#include <string>
#include <vector>
#include <algorithm>
#include <sstream>
#include <cctype>
#include <queue>

class InorderPostorder {
private:
    struct TreeNode {
        int val;
        TreeNode *left;
        TreeNode *right;
        TreeNode(int x) : val(x), left(NULL), right(NULL) {}
    };

    TreeNode* root;

    std::vector<int> parseNums(std::string str);
    TreeNode* buildTree(std::vector<int>& inorder, std::vector<int>& postorder);

public:
    InorderPostorder(const std::string& str);
    void printTree();
};

InorderPostorder::InorderPostorder(const std::string& str) : root(nullptr) {
    std::string cleaned = str;
    cleaned.erase(std::remove_if(cleaned.begin(), cleaned.end(), ::isspace), cleaned.end());

    std::string inorder_str;
    std::string postorder_str;

    size_t in_pos = cleaned.find("inorder=[");
    if (in_pos != std::string::npos) {
        size_t start = in_pos + 9;
        size_t end = cleaned.find("]", start);
        if (end != std::string::npos) {
            inorder_str = cleaned.substr(start, end - start);
        }
    }

    size_t post_pos = cleaned.find("postorder=[");
    if (post_pos != std::string::npos) {
        size_t start = post_pos + 11;
        size_t end = cleaned.find("]", start);
        if (end != std::string::npos) {
            postorder_str = cleaned.substr(start, end - start);
        }
    }

    std::vector<int> inorder = parseNums(inorder_str);
    std::vector<int> postorder = parseNums(postorder_str);

    root = buildTree(inorder, postorder);
}

std::vector<int> InorderPostorder::parseNums(std::string str) {
    std::vector<int> res;
    for (char &c : str) {
        if (c == ',') c = ' ';
    }
    std::stringstream ss(str);
    int val;
    while (ss >> val) {
        res.push_back(val);
    }
    return res;
}

InorderPostorder::TreeNode* InorderPostorder::buildTree(std::vector<int>& inorder, std::vector<int>& postorder) {
    if (inorder.empty() || postorder.empty()) {
        return nullptr;
    }

    int rootVal = postorder.back();
    postorder.pop_back();

    TreeNode* node = new TreeNode(rootVal);

    auto it = std::find(inorder.begin(), inorder.end(), rootVal);

    std::vector<int> left_inorder(inorder.begin(), it);
    std::vector<int> right_inorder(it + 1, inorder.end());

    node->right = buildTree(right_inorder, postorder);
    node->left = buildTree(left_inorder, postorder);

    return node;
}

void InorderPostorder::printTree() {
    if (root == nullptr) {
        std::cout << "[]" << std::endl;
        return;
    }

    // 使用 string 容器，因为要存 "null"
    std::vector<std::string> result;
    std::queue<TreeNode*> q;
    
    q.push(root);

    while (!q.empty()) {
        TreeNode* curr = q.front();
        q.pop();

        if (curr != nullptr) {
            result.push_back(std::to_string(curr->val));
            // 不管孩子是不是空，都推入队列，因为我们需要记录 "null"
            q.push(curr->left);
            q.push(curr->right);
        } else {
            result.push_back("null");
        }
    }

    // 【关键步骤】去除末尾多余的 "null"
    // 例如队列最后可能是 [..., 15, 7, null, null, null, null]
    // 我们需要把后面没用的 null 删掉，变成 [..., 15, 7]
    while (!result.empty() && result.back() == "null") {
        result.pop_back();
    }

    // 格式化输出
    std::cout << "[";
    for (size_t i = 0; i < result.size(); ++i) {
        std::cout << result[i];
        if (i != result.size() - 1) {
            std::cout << ",";
        }
    }
    std::cout << "]" << std::endl;
}

int main() {
    std::string line;
    if (std::getline(std::cin, line)) {
        InorderPostorder tree(line);
        tree.printTree();
    }
    return 0;
}
