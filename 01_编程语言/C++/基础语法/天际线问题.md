#include <iostream>
#include <vector>
#include <string>
#include <sstream>
#include <cctype>
#include <algorithm>
#include <set>

struct Building {
    int left;
    int right;
    int height;
    
    Building(int l, int r, int h) : left(l), right(r), height(h) {}
};

class Skyline {
private:
    std::vector<Building> _buildings;

    void _parse(const std::string& str);
    std::vector<std::vector<int>> _find_skyline();
public:
    Skyline(const std::string& str);
    
    void PrintSkyline();
};

Skyline::Skyline(const std::string& str) {
    _parse(str);
}
void Skyline::_parse(const std::string& str) {

    size_t startPos = str.find('[');
    if (startPos == std::string::npos) return;

    std::string buffer = str.substr(startPos);
    for (char &c : buffer) {
        if (!std::isdigit(c)) {
            c = ' ';
        }
    }

    std::stringstream ss(buffer);
    int l, r, h;

    while (ss >> l >> r >> h) {
        _buildings.emplace_back(l, r, h);
    }
}

std::vector<std::vector<int>> Skyline::_find_skyline() {
    std::vector<std::pair<int, int>> events;
    for (auto &building : _buildings) {
        events.emplace_back(building.left, -building.height);
        events.emplace_back(building.right, building.height);
    }
    std::sort(events.begin(), events.end());

    std::vector<std::vector<int>> skyline;
    std::multiset<int> heights;

    heights.insert(0);
    int prevHeight = 0;

    for (auto &event : events) { 
        int x = event.first;
        int h = event.second;

        if (h < 0) {
            heights.insert(-h);
        } else {
            auto it = heights.find(h);
            if(it != heights.end()) {
                heights.erase(it);
            }
        }

        int currentHeight = *heights.rbegin();
        if(currentHeight != prevHeight) {
            skyline.push_back({x, currentHeight});
            prevHeight = currentHeight;
        }
        
    }
    return skyline;
    
}

void Skyline::PrintSkyline() {
    auto skyline = _find_skyline();
    std::cout << "[";

    for (size_t i = 0; i < skyline.size(); ++i) {
        std::cout << "[" << skyline[i][0] << "," << skyline[i][1] << "]";
        if (i != skyline.size() - 1) {
            std::cout << ",";
        }
    }
    std::cout << "]" << std::endl;
}
int main() {
    std::string line;
    std::getline(std::cin, line);
    Skyline skyline(line);
    skyline.PrintSkyline();

    return 0;
}
