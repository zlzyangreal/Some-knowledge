#include <iostream>
#include <string>
#include <vector>
#include <algorithm>
#include <sstream>
#include <cctype>

class Employee {
private:
    struct Task {
        int id;
        int duration;
        
        Task(int task_id, int task_duration) 
            : id(task_id), duration(task_duration) {}
    };
    
    std::vector<Task> tasks_;
    int n_;

    void parseAndBuildTasks(const std::string& str);
    
public:
    explicit Employee(const std::string& str);
    int getMaxTimeTaskId() const;
};

void Employee::parseAndBuildTasks(const std::string& str) {
    // Remove whitespace
    std::string cleaned;
    cleaned.reserve(str.size());
    std::copy_if(str.begin(), str.end(), std::back_inserter(cleaned),
                 [](unsigned char c) { return !std::isspace(c); });

    // Parse n value
    const size_t n_pos = cleaned.find("n=");
    if (n_pos != std::string::npos) {
        const size_t comma_pos = cleaned.find(",", n_pos);
        n_ = std::stoi(cleaned.substr(n_pos + 2, comma_pos - n_pos - 2));
    }

    // Parse logs
    const size_t logs_start = cleaned.find("logs=[");
    const size_t logs_end = cleaned.find_last_of("]");
    
    if (logs_start == std::string::npos || logs_end == std::string::npos) {
        return;
    }

    std::string logs_str = cleaned.substr(logs_start + 6, logs_end - logs_start - 6);
    
    // Replace non-digits with spaces
    std::transform(logs_str.begin(), logs_str.end(), logs_str.begin(),
                   [](unsigned char c) { return std::isdigit(c) ? c : ' '; });

    // Parse task pairs and build tasks directly
    std::istringstream iss(logs_str);
    int task_id, end_time, prev_time = 0;
    
    while (iss >> task_id >> end_time) {
        tasks_.emplace_back(task_id, end_time - prev_time);
        prev_time = end_time;
    }
}

Employee::Employee(const std::string& str) : n_(0) {
    tasks_.reserve(10); // Reserve some space to avoid reallocations
    parseAndBuildTasks(str);
}

int Employee::getMaxTimeTaskId() const {
    if (tasks_.empty()) {
        return -1;
    }
    
    const auto max_task = std::max_element(tasks_.begin(), tasks_.end(),
        [](const Task& a, const Task& b) {
            return a.duration < b.duration;
        });
    
    return max_task->id;
}

int main() {
    std::string line;
    if (std::getline(std::cin, line)) {
        Employee emp(line);
        std::cout << emp.getMaxTimeTaskId() << '\n';
    }
    return 0;
}
